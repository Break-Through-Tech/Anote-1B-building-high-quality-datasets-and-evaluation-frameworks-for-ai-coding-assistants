import json
import sys


class LayoutConfigRecreation:
    def __init__(self):
        self.widgets = {}
        self.layout_tree = None

    def parse_config(self, config_str):
        try:
            data = json.loads(config_str)
        except json.JSONDecodeError:
            data = self._parse_custom_format(config_str)
        return data

    def _parse_custom_format(self, s):
        lines = s.strip().split('\n')
        result = {}
        stack = [result]
        for line in lines:
            line = line.rstrip()
            if not line.strip():
                continue
            indent = len(line) - len(line.lstrip())
            level = indent // 2
            content = line.strip()

            while len(stack) > level + 1:
                stack.pop()

            if ':' in content:
                key, _, value = content.partition(':')
                key = key.strip()
                value = value.strip()
                if value == '':
                    new_dict = {}
                    stack[-1][key] = new_dict
                    stack.append(new_dict)
                else:
                    stack[-1][key] = self._infer_type(value)
        return result

    def _infer_type(self, value):
        if value.lower() == 'true':
            return True
        if value.lower() == 'false':
            return False
        try:
            if '.' in value:
                return float(value)
            return int(value)
        except ValueError:
            return value.strip('"\'')

    def build_layout_tree(self, config):
        def build_node(node_data, node_id="root"):
            node = {
                "id": node_id,
                "type": node_data.get("type", "container"),
                "x": node_data.get("x", 0),
                "y": node_data.get("y", 0),
                "width": node_data.get("width", 0),
                "height": node_data.get("height", 0),
                "properties": node_data.get("properties", {}),
                "children": []
            }
            children = node_data.get("children", [])
            for idx, child in enumerate(children):
                child_id = child.get("id", f"{node_id}_child_{idx}")
                node["children"].append(build_node(child, child_id))
            return node

        self.layout_tree = build_node(config)
        return self.layout_tree

    def recreate_layout(self, tree=None):
        if tree is None:
            tree = self.layout_tree
        if tree is None:
            return None

        recreated = {
            "id": tree["id"],
            "type": tree["type"],
            "bounds": {
                "x": tree["x"],
                "y": tree["y"],
                "width": tree["width"],
                "height": tree["height"]
            },
            "properties": tree["properties"],
            "children": [self.recreate_layout(child) for child in tree["children"]]
        }
        return recreated

    def flatten_layout(self, tree=None, result=None):
        if result is None:
            result = []
        if tree is None:
            tree = self.layout_tree
        if tree is None:
            return result

        result.append({
            "id": tree["id"],
            "type": tree["type"],
            "x": tree["x"],
            "y": tree["y"],
            "width": tree["width"],
            "height": tree["height"]
        })

        for child in tree["children"]:
            self.flatten_layout(child, result)

        return result

    def to_json(self, tree=None, indent=2):
        if tree is None:
            tree = self.recreate_layout()
        return json.dumps(tree, indent=indent)

    def validate_layout(self, tree=None):
        if tree is None:
            tree = self.layout_tree
        if tree is None:
            return False, ["No layout tree found"]

        errors = []

        def check_node(node, path="root"):
            if node["width"] < 0:
                errors.append(f"{path}: negative width")
            if node["height"] < 0:
                errors.append(f"{path}: negative height")

            for i, child in enumerate(node["children"]):
                child_path = f"{path}/{child['id']}"
                if child["x"] < node["x"] or child["y"] < node["y"]:
                    errors.append(f"{child_path}: position outside parent bounds")

                child_right = child["x"] + child["width"]
                child_bottom = child["y"] + child["height"]
                parent_right = node["x"] + node["width"]
                parent_bottom = node["y"] + node["height"]

                if child_right > parent_right or child_bottom > parent_bottom:
                    errors.append(f"{child_path}: exceeds parent bounds")

                check_node(child, child_path)

        check_node(tree)
        return len(errors) == 0, errors

    def find_widget_by_id(self, widget_id, tree=None):
        if tree is None:
            tree = self.layout_tree
        if tree is None:
            return None

        if tree["id"] == widget_id:
            return tree

        for child in tree["children"]:
            result = self.find_widget_by_id(widget_id, child)
            if result:
                return result

        return None

    def update_widget(self, widget_id, updates):
        widget = self.find_widget_by_id(widget_id)
        if widget is None:
            return False

        for key, value in updates.items():
            if key in widget:
                widget[key] = value
            elif key in widget.get("properties", {}):
                widget["properties"][key] = value

        return True

    def calculate_total_area(self, tree=None):
        if tree is None:
            tree = self.layout_tree
        if tree is None:
            return 0

        total = tree["width"] * tree["height"]
        for child in tree["children"]:
            total += self.calculate_total_area(child)

        return total

    def get_widget_count(self, tree=None):
        if tree is None:
            tree = self.layout_tree
        if tree is None:
            return 0

        count = 1
        for child in tree["children"]:
            count += self.get_widget_count(child)

        return count

    def export_config(self, filepath, tree=None):
        if tree is None:
            tree = self.recreate_layout()

        with open(filepath, 'w') as f:
            json.dump(tree, f, indent=2)

    def import_config(self, filepath):
        with open(filepath, 'r') as f:
            content = f.read()

        config = self.parse_config(content)
        return self.build_layout_tree(config)


def main():
    if len(sys.argv) < 2:
        print("Usage: python layout_config_recreation.py <config_file>")
        sys.exit(1)

    filepath = sys.argv[1]
    recreator = LayoutConfigRecreation()

    try:
        tree = recreator.import_config(filepath)
        print("Layout tree built successfully")

        is_valid, errors = recreator.validate_layout()
        if is_valid:
            print("Layout is valid")
        else:
            print("Layout validation errors:")
            for error in errors:
                print(f"  - {error}")

        widget_count = recreator.get_widget_count()
        total_area = recreator.calculate_total_area()

        print(f"Total widgets: {widget_count}")
        print(f"Total area: {total_area}")

        recreated = recreator.recreate_layout()
        print("\nRecreated layout:")
        print(recreator.to_json(recreated))

    except FileNotFoundError:
        print(f"Error: File '{filepath}' not found")
        sys.exit(1)
    except Exception as e:
        print(f"Error processing config: {e}")
        sys.exit(1)


if __name__ == "__main__":
    main()

import sys

def main():
    data = sys.stdin.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    W = int(data[idx]); idx += 1
    H = int(data[idx]); idx += 1

    grid = [['.' for _ in range(W)] for _ in range(H)]

    for _ in range(n):
        comp_id = data[idx]; idx += 1
        x = int(data[idx]); idx += 1
        y = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        h = int(data[idx]); idx += 1

        ch = comp_id[0] if comp_id else '#'

        for row in range(y, min(y + h, H)):
            for col in range(x, min(x + w, W)):
                if 0 <= row < H and 0 <= col < W:
                    grid[row][col] = ch

    output = []
    for row in grid:
        output.append(''.join(row))

    print('\n'.join(output))

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
import os
import sys
import json
import shutil
import hashlib
import argparse
from datetime import datetime

class RSArchive:
    def __init__(self, archive_dir=".rs-archive"):
        self.archive_dir = archive_dir
        self.data_dir = os.path.join(archive_dir, "data")
        self.index_file = os.path.join(archive_dir, "index.json")
        self._ensure_dirs()

    def _ensure_dirs(self):
        os.makedirs(self.data_dir, exist_ok=True)
        if not os.path.exists(self.index_file):
            self._save_index({})

    def _load_index(self):
        if os.path.exists(self.index_file):
            with open(self.index_file, "r") as f:
                return json.load(f)
        return {}

    def _save_index(self, index):
        with open(self.index_file, "w") as f:
            json.dump(index, f, indent=2)

    def _hash_file(self, path):
        h = hashlib.sha256()
        with open(path, "rb") as f:
            for chunk in iter(lambda: f.read(8192), b""):
                h.update(chunk)
        return h.hexdigest()

    def add(self, filepath):
        if not os.path.isfile(filepath):
            print(f"Error: '{filepath}' is not a valid file.")
            return

        file_hash = self._hash_file(filepath)
        index = self._load_index()

        stored_name = file_hash
        stored_path = os.path.join(self.data_dir, stored_name)

        if not os.path.exists(stored_path):
            shutil.copy2(filepath, stored_path)

        entry = {
            "original_name": os.path.basename(filepath),
            "hash": file_hash,
            "size": os.path.getsize(filepath),
            "added_at": datetime.now().isoformat(),
        }

        index[file_hash] = entry
        self._save_index(index)
        print(f"Archived '{filepath}' as {file_hash[:12]}...")

    def list(self):
        index = self._load_index()
        if not index:
            print("Archive is empty.")
            return

        print(f"{'Hash':<14} {'Name':<30} {'Size':<10} {'Added At'}")
        print("-" * 80)
        for h, entry in index.items():
            print(f"{h[:12]:<14} {entry['original_name']:<30} {entry['size']:<10} {entry['added_at']}")

    def restore(self, file_hash, output_dir="."):
        index = self._load_index()
        matches = [h for h in index if h.startswith(file_hash)]

        if not matches:
            print(f"No archive entry found for hash '{file_hash}'")
            return
        if len(matches) > 1:
            print(f"Ambiguous hash '{file_hash}', matches: {matches}")
            return

        full_hash = matches[0]
        entry = index[full_hash]
        stored_path = os.path.join(self.data_dir, full_hash)
        output_path = os.path.join(output_dir, entry["original_name"])

        shutil.copy2(stored_path, output_path)
        print(f"Restored '{entry['original_name']}' to '{output_path}'")

    def remove(self, file_hash):
        index = self._load_index()
        matches = [h for h in index if h.startswith(file_hash)]

        if not matches:
            print(f"No archive entry found for hash '{file_hash}'")
            return
        if len(matches) > 1:
            print(f"Ambiguous hash '{file_hash}', matches: {matches}")
            return

        full_hash = matches[0]
        stored_path = os.path.join(self.data_dir, full_hash)

        if os.path.exists(stored_path):
            os.remove(stored_path)

        del index[full_hash]
        self._save_index(index)
        print(f"Removed archive entry '{file_hash}'")

    def info(self, file_hash):
        index = self._load_index()
        matches = [h for h in index if h.startswith(file_hash)]

        if not matches:
            print(f"No archive entry found for hash '{file_hash}'")
            return
        if len(matches) > 1:
            print(f"Ambiguous hash '{file_hash}', matches: {matches}")
            return

        full_hash = matches[0]
        entry = index[full_hash]
        print(json.dumps(entry, indent=2))

    def stats(self):
        index = self._load_index()
        total_size = sum(entry["size"] for entry in index.values())
        print(f"Total files: {len(index)}")
        print(f"Total size: {total_size} bytes")


def main():
    parser = argparse.ArgumentParser(description="rs-archive-clone: A simple content-addressable file archive.")
    subparsers = parser.add_subparsers(dest="command", required=True)

    add_parser = subparsers.add_parser("add", help="Add a file to the archive")
    add_parser.add_argument("filepath", help="Path to the file to archive")

    list_parser = subparsers.add_parser("list", help="List archived files")

    restore_parser = subparsers.add_parser("restore", help="Restore a file from the archive")
    restore_parser.add_argument("hash", help="Hash (or prefix) of the file to restore")
    restore_parser.add_argument("-o", "--output", default=".", help="Output directory")

    remove_parser = subparsers.add_parser("remove", help="Remove a file from the archive")
    remove_parser.add_argument("hash", help="Hash (or prefix) of the file to remove")

    info_parser = subparsers.add_parser("info", help="Show info about an archived file")
    info_parser.add_argument("hash", help="Hash (or prefix) of the file")

    stats_parser = subparsers.add_parser("stats", help="Show archive statistics")

    args = parser.parse_args()
    archive = RSArchive()

    if args.command == "add":
        archive.add(args.filepath)
    elif args.command == "list":
        archive.list()
    elif args.command == "restore":
        archive.restore(args.hash, args.output)
    elif args.command == "remove":
        archive.remove(args.hash)
    elif args.command == "info":
        archive.info(args.hash)
    elif args.command == "stats":
        archive.stats()


if __name__ == "__main__":
    main()

import time
import threading
from collections import deque


class TokenBucket:
    """
    Token bucket rate limiter to handle bursts of requests
    (e.g. for sglang / Qwen inference server throttling).
    """

    def __init__(self, rate: float, capacity: int):
        """
        :param rate: tokens added per second (sustained request rate)
        :param capacity: maximum burst size (max tokens in bucket)
        """
        self.rate = rate
        self.capacity = capacity
        self.tokens = capacity
        self.timestamp = time.monotonic()
        self.lock = threading.Lock()

    def _refill(self):
        now = time.monotonic()
        elapsed = now - self.timestamp
        self.timestamp = now
        self.tokens = min(self.capacity, self.tokens + elapsed * self.rate)

    def acquire(self, tokens: int = 1) -> bool:
        with self.lock:
            self._refill()
            if self.tokens >= tokens:
                self.tokens -= tokens
                return True
            return False

    def wait_for_token(self, tokens: int = 1):
        while True:
            with self.lock:
                self._refill()
                if self.tokens >= tokens:
                    self.tokens -= tokens
                    return
                deficit = tokens - self.tokens
                sleep_time = deficit / self.rate
            time.sleep(sleep_time)


class BurstRequestQueue:
    """
    Queue that groups incoming requests into bursts, respecting
    a max batch size and max wait time, useful for batching
    requests to an sglang / Qwen serving backend.
    """

    def __init__(self, max_batch_size: int = 32, max_wait_time: float = 0.05):
        self.max_batch_size = max_batch_size
        self.max_wait_time = max_wait_time
        self.queue = deque()
        self.lock = threading.Lock()
        self.condition = threading.Condition(self.lock)

    def submit(self, request):
        with self.lock:
            self.queue.append(request)
            self.condition.notify_all()

    def get_batch(self):
        with self.condition:
            start_time = time.monotonic()
            while True:
                if len(self.queue) >= self.max_batch_size:
                    break
                elapsed = time.monotonic() - start_time
                remaining = self.max_wait_time - elapsed
                if remaining <= 0:
                    break
                self.condition.wait(timeout=remaining)

            batch = []
            while self.queue and len(batch) < self.max_batch_size:
                batch.append(self.queue.popleft())
            return batch


class SGLangQwenBurstScheduler:
    """
    Combines rate limiting and request batching to handle
    burst traffic to an sglang-backed Qwen inference endpoint.
    """

    def __init__(self, rate: float, capacity: int,
                 max_batch_size: int = 32, max_wait_time: float = 0.05):
        self.bucket = TokenBucket(rate, capacity)
        self.queue = BurstRequestQueue(max_batch_size, max_wait_time)

    def submit_request(self, request):
        self.bucket.wait_for_token()
        self.queue.submit(request)

    def process_batches(self, handler):
        """
        Continuously pull batches from the queue and process
        them with the provided handler function.
        """
        while True:
            batch = self.queue.get_batch()
            if batch:
                handler(batch)
            else:
                time.sleep(0.001)


if __name__ == "__main__":
    def dummy_handler(batch):
        print(f"Processing batch of {len(batch)} requests: {batch}")

    scheduler = SGLangQwenBurstScheduler(rate=10, capacity=20,
                                          max_batch_size=8, max_wait_time=0.1)

    worker = threading.Thread(target=scheduler.process_batches, args=(dummy_handler,), daemon=True)
    worker.start()

    for i in range(50):
        scheduler.submit_request(f"req-{i}")
        time.sleep(0.01)

    time.sleep(1)
